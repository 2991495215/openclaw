import { createDeferred as defer, createEffect, onSettled } from "solid-js";

export function validLifecycle(read: () => string) {
  createEffect(read, (value) => {
    document.title = value;
  });
  onSettled(() => {
    const listener = () => {};
    document.addEventListener("example", listener);
    return () => document.removeEventListener("example", listener);
  });
}

export function validDeferred(read: () => string) {
  return defer(read);
}
